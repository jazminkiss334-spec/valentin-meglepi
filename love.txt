<script>
const stages = [
    { q: "> Enter system vendor:", a: "samsung" },
    { q: "> Enter relationship initialization date (MM-DD):", a: "03-08" },
    { q: "> First kiss location? (Budapest riverbank)", a: "dunapart" },
    { q: "> Pre-intimacy carbohydrate protocol?", a: "spagetti" },
    { q: "> Her favorite flower?", a: "rozsa" },
    { q: "> Who loves more? (justice mode)", a: ["both","equal"] }
];

let currentStage = 0;
let attempts = 3;

const terminal = document.getElementById("terminal");

function typeLine(text, delay = 30) {
    return new Promise(resolve => {
        let i = 0;
        const interval = setInterval(() => {
            terminal.innerHTML += text.charAt(i);
            i++;
            if (i >= text.length) {
                clearInterval(interval);
                terminal.innerHTML += "<br>";
                resolve();
            }
        }, delay);
    });
}

async function bootSequence() {
    await typeLine("Booting GALAXY_PAIRING_PROTOCOL...");
    await typeLine("Initializing Samsung Secure Heart v2.5...");
    await typeLine("Loading encrypted memories...");
    await typeLine("Access level: RESTRICTED");
    await typeLine("");
    await typeLine(stages[currentStage].q);
    document.getElementById("inputArea").classList.remove("hidden");
}

function submitAnswer() {
    const input = document.getElementById("userInput");
    let value = input.value.toLowerCase().trim();
    input.value = "";

    let correct = stages[currentStage].a;

    if (Array.isArray(correct) ? correct.includes(value) : value === correct) {
        nextStage();
    } else {
        attempts--;
        terminal.innerHTML += "✖ Incorrect. Attempts left: " + attempts + "<br>";
        if (attempts === 0) {
            terminal.innerHTML += "<span class='locked'>SYSTEM LOCKED.</span>";
            document.getElementById("inputArea").classList.add("hidden");
        }
    }
}

async function nextStage() {
    terminal.innerHTML += "✔ Access granted.<br><br>";
    currentStage++;
    attempts = 3;

    if (currentStage < stages.length) {
        await typeLine(stages[currentStage].q);
    } else {
        startFinal();
    }
}

async function startFinal() {
    document.getElementById("inputArea").classList.add("hidden");

    await typeLine("Decrypting emotional core...");
    await typeLine("Running unit tests...");

    const tests = [
        "✔ Loyalty",
        "✔ Passion",
        "✔ Cooking skills",
        "✔ Matching energy",
        "✔ Justice system integrity",
        "✔ Infinite obsession mode"
    ];

    for (let test of tests) {
        await typeLine(test, 20);
    }

    await typeLine("");
    await typeLine("Finalizing...");
    await new Promise(r => setTimeout(r, 1500));

    document.getElementById("finalSection").classList.remove("hidden");
}

bootSequence();
